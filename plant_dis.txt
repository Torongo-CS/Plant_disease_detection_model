# ============================================================
# PREPROCESSING
# ============================================================

def preprocess_image(
    rgb_image,
    input_details
):
    """
    Exact training preprocessing:

        RGB
        ↓
        resize 224x224
        ↓
        float32
        ↓
        pixel / 255.0
        ↓
        [1, 224, 224, 3]

    Quantized models are handled using their
    TFLite quantization parameters.
    """

    input_info = input_details[0]

    input_shape = tuple(
        input_info["shape"]
    )

    input_dtype = input_info["dtype"]


    # --------------------------------------------------------
    # Resize
    # --------------------------------------------------------

    image = cv2.resize(
        rgb_image,
        (
            IMAGE_SIZE,
            IMAGE_SIZE
        ),
        interpolation=cv2.INTER_AREA
    )


    # --------------------------------------------------------
    # IMPORTANT:
    #
    # The image is ALREADY RGB.
    #
    # DO NOT perform:
    #
    # cv2.cvtColor(
    #     image,
    #     cv2.COLOR_BGR2RGB
    # )
    #
    # here.
    # --------------------------------------------------------

    image = image.astype(
        np.float32
    )


    # --------------------------------------------------------
    # EXACT TRAINING NORMALIZATION
    # --------------------------------------------------------

    image = image / 255.0


    # --------------------------------------------------------
    # Add batch dimension
    # --------------------------------------------------------

    image = np.expand_dims(
        image,
        axis=0
    )


    # --------------------------------------------------------
    # FLOAT32 MODEL
    # --------------------------------------------------------

    if input_dtype == np.float32:

        tensor = image.astype(
            np.float32
        )


    # --------------------------------------------------------
    # UINT8 MODEL
    # --------------------------------------------------------

    elif input_dtype == np.uint8:

        scale, zero_point = (
            input_info["quantization"]
        )

        if scale == 0:

            raise RuntimeError(
                "Invalid uint8 quantization scale."
            )

        tensor = np.round(
            image / scale
            + zero_point
        )

        tensor = np.clip(
            tensor,
            0,
            255
        ).astype(
            np.uint8
        )


    # --------------------------------------------------------
    # INT8 MODEL
    # --------------------------------------------------------

    elif input_dtype == np.int8:

        scale, zero_point = (
            input_info["quantization"]
        )

        if scale == 0:

            raise RuntimeError(
                "Invalid int8 quantization scale."
            )

        tensor = np.round(
            image / scale
            + zero_point
        )

        tensor = np.clip(
            tensor,
            -128,
            127
        ).astype(
            np.int8
        )


    else:

        raise RuntimeError(
            f"Unsupported input dtype: "
            f"{input_dtype}"
        )


    # --------------------------------------------------------
    # Shape validation
    # --------------------------------------------------------

    if tuple(tensor.shape) != input_shape:

        raise RuntimeError(
            f"Input shape mismatch. "
            f"Expected {input_shape}, "
            f"got {tensor.shape}."
        )


    return tensor


# ============================================================
# OUTPUT DEQUANTIZATION
# ============================================================

def dequantize_output(
    output,
    output_details
):
    """
    Converts quantized TFLite output back to
    floating-point values.
    """

    output_info = output_details[0]

    dtype = output_info["dtype"]


    if dtype in (
        np.uint8,
        np.int8
    ):

        scale, zero_point = (
            output_info["quantization"]
        )

        if scale != 0:

            output = (
                output.astype(
                    np.float32
                )
                - zero_point
            ) * scale

        else:

            output = output.astype(
                np.float32
            )

    else:

        output = output.astype(
            np.float32
        )


    return output


# ============================================================
# SOFTMAX
# ============================================================

def softmax(values):

    values = np.asarray(
        values,
        dtype=np.float32
    )

    values = (
        values
        - np.max(values)
    )

    exponentials = np.exp(
        values
    )

    total = np.sum(
        exponentials
    )

    if total <= 0:

        raise RuntimeError(
            "Softmax produced invalid output."
        )

    return (
        exponentials / total
    )


# ============================================================
# OUTPUT -> PROBABILITIES
# ============================================================

def output_to_probabilities(
    output
):
    """
    Handles both:

    A) Models whose final layer is softmax:

       [0.01, 0.02, 0.91, ...]

    B) Models whose output is logits:

       [-2.1, 0.3, 5.7, ...]

    Softmax is ONLY applied when the output does not
    already look like a probability distribution.
    """

    output = np.asarray(
        output,
        dtype=np.float32
    )

    output = np.squeeze(
        output
    )

    output = output.reshape(
        -1
    )


    # --------------------------------------------------------
    # Validate number of classes
    # --------------------------------------------------------

    if len(output) != len(
        CLASS_NAMES
    ):

        raise RuntimeError(
            f"Model returned "
            f"{len(output)} outputs. "
            f"Expected "
            f"{len(CLASS_NAMES)}."
        )


    # --------------------------------------------------------
    # Validate values
    # --------------------------------------------------------

    if not np.all(
        np.isfinite(output)
    ):

        raise RuntimeError(
            "Model produced NaN "
            "or infinite values."
        )


    # --------------------------------------------------------
    # Determine whether output already
    # represents probabilities.
    #
    # A softmax output should:
    #
    #   0 <= p <= 1
    #   sum(p) ≈ 1
    # --------------------------------------------------------

    output_sum = float(
        np.sum(output)
    )

    looks_like_probability = (
        np.all(output >= 0.0)
        and
        np.all(output <= 1.0)
        and
        abs(
            output_sum - 1.0
        ) < 0.05
    )


    if looks_like_probability:

        probabilities = output.copy()

    else:

        # Treat output as logits.
        probabilities = softmax(
            output
        )


    # --------------------------------------------------------
    # Final normalization
    # --------------------------------------------------------

    probabilities = np.clip(
        probabilities,
        0.0,
        1.0
    )

    probability_sum = np.sum(
        probabilities
    )

    if probability_sum <= 0:

        raise RuntimeError(
            "Invalid probability distribution."
        )

    probabilities = (
        probabilities
        / probability_sum
    )


    return probabilities.astype(
        np.float32
    )


# ============================================================
# RUN ONE MODEL
# ============================================================

def run_model(
    interpreter,
    input_details,
    output_details,
    rgb_image
):

    # --------------------------------------------------------
    # PREPROCESS
    # --------------------------------------------------------

    input_tensor = preprocess_image(
        rgb_image,
        input_details
    )


    # --------------------------------------------------------
    # INPUT
    # --------------------------------------------------------

    interpreter.set_tensor(
        input_details[0]["index"],
        input_tensor
    )


    # --------------------------------------------------------
    # INFERENCE
    # --------------------------------------------------------

    interpreter.invoke()


    # --------------------------------------------------------
    # RAW OUTPUT
    # --------------------------------------------------------

    raw_output = interpreter.get_tensor(
        output_details[0]["index"]
    )


    # --------------------------------------------------------
    # DEQUANTIZE
    # --------------------------------------------------------

    raw_output = dequantize_output(
        raw_output,
        output_details
    )


    # --------------------------------------------------------
    # PROBABILITIES
    # --------------------------------------------------------

    probabilities = (
        output_to_probabilities(
            raw_output
        )
    )


    return probabilities


# ============================================================
# ENSEMBLE
# ============================================================

def ensemble_prediction(
    rgb_image
):

    # ========================================================
    # MODEL 1
    # MobileNetV3 Large
    # Reported accuracy: 94%
    # ========================================================

    with model1_lock:

        probs1 = run_model(
            model1,
            model1_input_details,
            model1_output_details,
            rgb_image
        )


    # ========================================================
    # MODEL 2
    # pd_v1
    # Reported accuracy: 68%
    # ========================================================

    with model2_lock:

        probs2 = run_model(
            model2,
            model2_input_details,
            model2_output_details,
            rgb_image
        )


    # ========================================================
    # WEIGHTED ENSEMBLE
    #
    # MobileNetV3 = 80%
    # pd_v1       = 20%
    #
    # This prevents the weaker model from overpowering
    # the stronger model simply because it has a higher
    # raw confidence.
    # ========================================================

    ensemble_probs = (
        MODEL_1_WEIGHT * probs1
        +
        MODEL_2_WEIGHT * probs2
    )


    # Normalize
    ensemble_probs = (
        ensemble_probs
        / np.sum(ensemble_probs)
    )


    # ========================================================
    # FINAL PREDICTION
    # ========================================================

    predicted_index = int(
        np.argmax(
            ensemble_probs
        )
    )

    predicted_class = (
        CLASS_NAMES[
            predicted_index
        ]
    )

    final_confidence = float(
        ensemble_probs[
            predicted_index
        ]
    )


    # ========================================================
    # INDIVIDUAL MODEL RESULTS
    # ========================================================

    model1_index = int(
        np.argmax(probs1)
    )

    model2_index = int(
        np.argmax(probs2)
    )


    return {

        "class":
            predicted_class,

        "index":
            predicted_index,

        "confidence":
            final_confidence,

        "ensemble":
            ensemble_probs.tolist(),

        "model1":
            probs1.tolist(),

        "model2":
            probs2.tolist(),

        "model1_prediction":
            CLASS_NAMES[
                model1_index
            ],

        "model1_confidence":
            float(
                probs1[
                    model1_index
                ]
            ),

        "model2_prediction":
            CLASS_NAMES[
                model2_index
            ],

        "model2_confidence":
            float(
                probs2[
                    model2_index
                ]
            )
    }

def get_ai_advice(
    disease,
    confidence
):

    if not OPENROUTER_API_KEY:

        return (
            "AI advice is currently unavailable.\n\n"
            "Set the OPENROUTER_API_KEY environment variable "
            "to enable the AI plant-care assistant."
        )


    try:

        from openai import OpenAI


        client = OpenAI(

            base_url=(
                "https://openrouter.ai/api/v1"
            ),

            api_key=(
                OPENROUTER_API_KEY
            )
        )


        prompt = f"""
You are a plant-care assistant.

A Raspberry Pi plant disease detection system produced
the following prediction:

Detected condition: {disease}
Model confidence: {confidence:.1f}%

Give concise, practical advice for a normal plant owner.

Explain:

1. What the detected condition means.
2. What they should do now.
3. How they can reduce the chance of spreading it.
4. General environmental practices that may help.

Keep the answer easy to understand.

Do not claim the computer vision prediction is guaranteed
to be correct.

Do not prescribe dangerous chemicals or provide unsafe
pesticide dosages.

Use short sections and bullet points.
"""


        response = client.chat.completions.create(

            model=OPENROUTER_MODEL,

            messages=[

                {
                    "role": "system",

                    "content":
                        "You are a concise agricultural "
                        "plant-care assistant."
                },

                {
                    "role": "user",

                    "content":
                        prompt
                }
            ],

            temperature=0.2,

            max_tokens=500
        )


        return (
            response
            .choices[0]
            .message
            .content
        )


    except Exception as e:

        print(
            "OpenRouter error:",
            e
        )

        return (
            "The AI plant-care service could not be reached.\n\n"
            "Please check your OpenRouter API key and "
            "internet connection."
        )

# CAMERA MJPEG STREAM
# ============================================================

def generate_camera_stream():

    while True:

        frame = get_current_frame()


        if frame is None:

            time.sleep(
                0.05
            )

            continue


        # ----------------------------------------------------
        # CAMERA FRAME IS RGB.
        #
        # OpenCV's JPEG encoder expects BGR.
        #
        # Therefore ONLY convert the copy used for
        # browser JPEG encoding.
        #
        # The actual model input remains RGB.
        # ----------------------------------------------------

        bgr_frame = cv2.cvtColor(
            frame,
            cv2.COLOR_RGB2BGR
        )


        success, encoded = (
            cv2.imencode(
                ".jpg",
                bgr_frame,
                [
                    int(
                        cv2.IMWRITE_JPEG_QUALITY
                    ),
                    JPEG_QUALITY
                ]
            )
        )


        if not success:

            continue


        jpeg_bytes = (
            encoded.tobytes()
        )


        yield (
            b"--frame\r\n"
            b"Content-Type: image/jpeg\r\n"
            b"Cache-Control: no-cache\r\n"
            b"Pragma: no-cache\r\n\r\n"
            + jpeg_bytes
            + b"\r\n"
        )


        # Prevent excessive CPU usage.
        time.sleep(
            1.0 / CAMERA_FPS
        )
